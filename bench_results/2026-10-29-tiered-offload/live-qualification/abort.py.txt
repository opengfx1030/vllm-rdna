import concurrent.futures, http.client, json, re, socket, threading, time, urllib.request
from pathlib import Path
root=Path('/home/chenco_adm/w4a8_runs/moe-offload/tiered-first-pass/live-qualification')
log=Path('/home/chenco_adm/w4a8_runs/moe-offload/tiered-first-pass/live-qualification.log')
start=log.stat().st_size
conns={}; responses={}; lock=threading.Lock(); report={'scope':'disconnect after observed host-ready spill; resume may race disconnect','aborted':None}
def run(i):
    key=f'live-abort-{i}'
    c=http.client.HTTPConnection('127.0.0.1',18766,timeout=240)
    with lock: conns[key]=c
    body={'model':'flash-next','request_id':key,'prompt':f'Write a detailed numbered explanation of how a compiler works, with at least twenty examples. Case {i}:','max_tokens':256,'temperature':0,'ignore_eos':True,'stream':True}
    try:
        c.request('POST','/v1/completions',json.dumps(body),{'Content-Type':'application/json'})
        r=c.getresponse()
        with lock: responses[key]=r
        chunks=0
        while r.readline(): chunks+=1
        return {'id':key,'status':r.status,'lines':chunks}
    except Exception as e: return {'id':key,'error':repr(e)}
    finally: c.close()
with concurrent.futures.ThreadPoolExecutor(max_workers=9) as pool:
    fs=[pool.submit(run,i) for i in range(9)]
    deadline=time.monotonic()+240
    with log.open() as f:
        f.seek(start)
        while time.monotonic()<deadline and not all(x.done() for x in fs):
            text=f.read()
            m=re.search(r'Live KV spill cmpl-(live-abort-\d+)',text)
            if m:
                key=m[1]
                with lock:
                    c=conns.get(key); r=responses.get(key)
                    s=c.sock if c else None
                    if s is None and r is not None and r.fp is not None: s=r.fp.raw._sock
                if s is not None:
                    s.shutdown(socket.SHUT_RDWR)
                    report['aborted']={'id':key,'trigger':m[0],'monotonic':time.monotonic()}
                    break
            time.sleep(.01)
    report['requests']=[x.result(timeout=250) for x in fs]
report['passed']=report['aborted'] is not None and all(x.get('status')==200 or x['id']==report['aborted']['id'] for x in report['requests'])
(root/'abort.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps(report,indent=2),flush=True)
raise SystemExit(0 if report['passed'] else 1)
