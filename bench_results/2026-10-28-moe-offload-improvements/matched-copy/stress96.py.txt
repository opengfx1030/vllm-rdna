 spec=importlib.util.spec_from_file_location(name,path);module=importlib.util.module_from_spec(spec);sys.modules[name]=module;spec.loader.exec_module(module)
sizes=[1638400,51200,12800,819200,25600,6400]
dtypes=[torch.int32,torch.float16,torch.int32,torch.int32,torch.float16,torch.int32]
bank={}
for p,(name,size,dtype) in enumerate(zip(module.PAYLOAD_NAMES,sizes,dtypes)):
 width=size//torch.empty((),dtype=dtype).element_size()
 source=torch.empty((96,width),dtype=dtype,pin_memory=True)
 pattern=torch.arange(width,dtype=torch.int32).remainder_(251)
 for expert in range(96):source[expert].copy_(pattern);source[expert].add_(expert*3+p)
 del pattern
 bank[name]=source

assert cache.host_bytes+cache.device_bytes<500_000_000
bindings=[(95-i,(i*5+3)%96,True) for i in range(96)]

 for tensor in cache.payload.values():tensor.fill_(-17)
 cache._copy(bindings)
 for tensor in cache.payload.values():tensor.add_(1)
cache._copy_stream.synchronize()
for expert,slot,_ in bindings:
 for name in module.PAYLOAD_NAMES:
  actual=cache.payload[name][slot].cpu().view(torch.uint8)
  expected=(cache.host_payload[name][expert]+1).view(torch.uint8)
  assert torch.equal(actual,expected),(name,expert,slot)
