#!/usr/bin/env python3
"""Matched actual PackedExpertCache._copy timings; requires explicit --run.

Use the same persistent pinned bank, GPU pool and nondefault stream for both
arms. GPU-event intervals include host submission gaps; these are not isolated
DMA-engine timings or model-level performance predictions.
"""
import argparse
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import statistics
import sys
import time


def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument('--run', action='store_true', help='explicitly authorize GPU8 copies')
    p.add_argument('--root', type=Path, default=Path('/home/chenco_adm/w4a8_runs/moe-offload/batch-copy-probe/native'))
    p.add_argument('--cycles', type=int, default=15, help='ABBA cycles, giving 2*cycles samples per arm')
    p.add_argument('--warmup-cycles', type=int, default=3)
    p.add_argument('--output', type=Path)
    args = p.parse_args()
    if not args.run:
        print('Prepared only: no torch import, GPU allocation, copies or timing. Add --run only after explicit scheduling signal.')
        return
    assert args.cycles >= 1 and args.warmup_cycles >= 1
    required = {'HIP_VISIBLE_DEVICES': '8', 'HSA_ENABLE_SDMA': '0', 'HSA_FORCE_FINE_GRAIN_PCIE': '1', 'GPU_MAX_HW_QUEUES': '4'}
    for name, value in required.items():
        assert os.environ.get(name) == value, (name, os.environ.get(name), value)
    assert not os.environ.get('ROCR_VISIBLE_DEVICES')
    assert os.environ.get('CUDA_VISIBLE_DEVICES') in (None, '', '8')  # venv sitecustomize mirrors HIP visibility
    import torch
    torch.cuda.set_device(0)
    assert torch.cuda.device_count() == 1
    assert torch.cuda.get_device_properties(0).gcnArchName.startswith('gfx103')
    native_library = args.root / 'native_copy.so'
    if hasattr(torch.ops._rocm_C, 'rdna2_expert_cache_copy'):
        native_library = Path('/home/chenco_adm/opengfx1030_vllm-rdna_moe_offloading/vllm/_rocm_C.abi3.so')
    else:
        torch.ops.load_library(str(native_library))
    directory = args.root / 'project/vllm/model_executor/layers/fused_moe'
    for name, path in (
        ('vllm.model_executor.layers.fused_moe.rdna_expert_cache_policy', directory / 'rdna_expert_cache_policy.py'),
        ('balanced_cache_copy_module', directory / 'rdna_expert_cache.py'),
    ):
        spec = importlib.util.spec_from_file_location(name, path)
        module = importlib.util.module_from_spec(spec)
        sys.modules[name] = module
        spec.loader.exec_module(module)
    sizes = [1638400, 51200, 12800, 819200, 25600, 6400]
    dtypes = [torch.int32, torch.float16, torch.int32, torch.int32, torch.float16, torch.int32]
    bank = {}
    for payload, size, dtype in zip(module.PAYLOAD_NAMES, sizes, dtypes, strict=True):
        width = size // torch.empty((), dtype=dtype).element_size()
        tensor = torch.empty((16, width), dtype=dtype, pin_memory=True)
        for expert in range(16):
            tensor[expert].fill_(expert * 13 + len(bank) + 1)
        bank[payload] = tensor
    cache = module.PackedExpertCache(bank, 16, 'cuda', copy_mode='native')
    native_op = cache._copy_op
    stream = cache._copy_stream
    pointers = [t.data_ptr() for t in cache._copy_sources + cache._copy_destinations]
    assert cache.host_bytes + cache.device_bytes < 82_000_000
    with torch.cuda.stream(stream):
        for tensor in cache.payload.values():
            tensor.fill_(-1)
    stream.synchronize()
    start = torch.cuda.Event(enable_timing=True)
    end = torch.cuda.Event(enable_timing=True)
    def verify(bindings):
        stream.synchronize()
        for expert, slot, _ in bindings:
            for name in module.PAYLOAD_NAMES:
                actual = cache.payload[name][slot].cpu().view(torch.uint8)
                expected = cache.host_payload[name][expert].view(torch.uint8)
                assert torch.equal(actual, expected), (name, expert, slot)
        assert pointers == [t.data_ptr() for t in cache._copy_sources + cache._copy_destinations]
    def one(mode, bindings):
        cache._copy_op = native_op if mode == 'native' else None
        with torch.cuda.stream(stream):
            start.record(stream)
            wall_start = time.perf_counter_ns()
            cache._copy(bindings)
            enqueue_end = time.perf_counter_ns()
            end.record(stream)
        end.synchronize()
        wall_end = time.perf_counter_ns()
        return {'cpu_enqueue_us': (enqueue_end-wall_start)/1000,
                'gpu_event_interval_us': start.elapsed_time(end)*1000,
                'wall_completion_us': (wall_end-wall_start)/1000}
    def summarize(values):
        ordered = sorted(values)
        def percentile(q):
            at = (len(ordered)-1)*q
            lo = int(at)
            hi = min(lo+1, len(ordered)-1)
            return ordered[lo] + (ordered[hi]-ordered[lo])*(at-lo)
        return {'median': statistics.median(values), 'p10': percentile(.1), 'p90': percentile(.9)}
    result = {'torch': torch.__version__, 'hip': torch.version.hip,
              'device': torch.cuda.get_device_name(0), 'env': required,
              'cache_sha256': hashlib.sha256((directory/'rdna_expert_cache.py').read_bytes()).hexdigest(),
              'native_library': str(native_library),
              'native_so_sha256': hashlib.sha256(native_library.read_bytes()).hexdigest(),
              'method': 'actual PackedExpertCache._copy including Python arguments and C++ descriptor construction',
              'order': ['tensor', 'native', 'native', 'tensor'],
              'warmup_cycles': args.warmup_cycles, 'measured_cycles': args.cycles,
              'payload_row_bytes': sizes, 'host_plus_gpu_bytes': cache.host_bytes+cache.device_bytes,
              'event_caveat': 'GPU-event interval includes CPU submission gaps; wall starts immediately before _copy and includes event recording/wait.',
              'cells': []}
    for misses in [1, 4, 16]:
        bindings = [((i*5+3)%16, (i*7+1)%16, True) for i in range(misses)]
        for mode in result['order'] * args.warmup_cycles:
            one(mode, bindings)
            verify(bindings)  # outside all timed intervals, both arms checked
        counters_before = (cache.native_copy_calls, cache.batch_copy_calls, cache.copied_bytes)
        samples = []
        for cycle in range(args.cycles):
            for arm, mode in enumerate(result['order']):
                sample = one(mode, bindings)
                sample.update(mode=mode, cycle=cycle, arm=arm)
                samples.append(sample)
        verify(bindings)
        counters_after = (cache.native_copy_calls, cache.batch_copy_calls, cache.copied_bytes)
        assert counters_after[0]-counters_before[0] == 2*args.cycles
        assert counters_after[1]-counters_before[1] == 2*args.cycles
        assert counters_after[2]-counters_before[2] == 4*args.cycles*misses*sum(sizes)
        cell = {'misses': misses, 'bytes': misses*sum(sizes), 'bindings': bindings,
                'exact_bytes': True, 'samples': samples, 'summary': {}}
        for mode in ['tensor', 'native']:
            selected = [s for s in samples if s['mode']==mode]
            cell['summary'][mode] = {key: summarize([s[key] for s in selected])
                                    for key in ['cpu_enqueue_us','gpu_event_interval_us','wall_completion_us']}
        result['cells'].append(cell)
        print(json.dumps({'misses': misses, 'summary': cell['summary']}), flush=True)
    if args.output:
        args.output.write_text(json.dumps(result, indent=2)+'\n')
    else:
        print(json.dumps(result), flush=True)

if __name__ == '__main__':
    main()
