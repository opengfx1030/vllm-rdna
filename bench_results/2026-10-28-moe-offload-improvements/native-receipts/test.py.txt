import json, os
import torch
root='/home/chenco_adm/w4a8_runs/moe-offload/batch-copy-probe/native'
assert os.environ['HIP_VISIBLE_DEVICES']=='8'
torch.cuda.set_device(0)
torch.ops.load_library(root+'/native_copy.so')
op=torch.ops._rocm_C.rdna2_expert_cache_copy
sizes=[1638400,51200,12800,819200,25600,6400]
dtypes=[torch.int32,torch.float16,torch.int32,torch.int32,torch.float16,torch.int32]
sources=[];destinations=[]
for p,(size,dtype) in enumerate(zip(sizes,dtypes)):
 n=size//torch.empty((),dtype=dtype).element_size()
 s=torch.empty((5,n),dtype=dtype,pin_memory=True)
 for e in range(5):s[e].fill_(10*p+e)
 sources.append(s)
 destinations.append(torch.empty((4,n),dtype=dtype,device='cuda'))
stream=torch.cuda.Stream()
def exact(expected):
 stream.synchronize()
 return all(torch.equal(d.cpu().view(torch.uint8),e.view(torch.uint8)) for d,e in zip(destinations,expected))
proof={'device':torch.cuda.get_device_name(0),'arch':torch.cuda.get_device_properties(0).gcnArchName,'torch':torch.__version__,'hip':torch.version.hip,'payload_bytes_host_plus_gpu':sum(sizes)*9}
for experts,slots in [([4,1,3],[0,3,1]),([2,0],[2,0])]:
 expected=[torch.full(d.shape,-4,dtype=d.dtype) for d in destinations]
 for p in range(6):
  for e,slot in zip(experts,slots):expected[p][slot].copy_(sources[p][e]+1)
 with torch.cuda.stream(stream):
  for d in destinations:d.fill_(-5)
  used=op(sources,destinations,experts,slots)
  for d in destinations:d.add_(1)
 assert used is True and exact(expected)
proof['batch_used_and_ordering_exact']=True
proof['invalid_inputs']={}
for name,src,experts,slots in [
 ('bad_last_expert',sources,[1,5],[0,1]),
 ('bad_last_slot',sources,[1,2],[0,4]),
 ('duplicate_slots',sources,[1,2],[0,0]),
 ('bad_final_payload_shape',sources[:-1]+[sources[-1][:,:-1].contiguous().pin_memory()],[1,2],[0,1])]:
 with torch.cuda.stream(stream):
  for d in destinations:d.fill_(-9)
 stream.synchronize()
 try:
  with torch.cuda.stream(stream):op(src,destinations,experts,slots)
 except RuntimeError as e:message=str(e).splitlines()[0]
 else:raise AssertionError(name+' accepted')
 assert exact([torch.full(d.shape,-9,dtype=d.dtype) for d in destinations])
 proof['invalid_inputs'][name]={'rejected':message,'no_writes':True}
with torch.cuda.stream(stream):assert op(sources,destinations,[],[]) is False
stream.synchronize()
graph=torch.cuda.CUDAGraph()
with torch.cuda.stream(stream):
 graph.capture_begin()
 try:op(sources,destinations,[0],[0])
 except RuntimeError as e:message=str(e).splitlines()[0]
 else:raise AssertionError('capture accepted')
 finally:graph.capture_end()
assert 'capture' in message
assert exact([torch.full(d.shape,-9,dtype=d.dtype) for d in destinations])
proof['raw_graph_capture']={'rejected':message,'no_writes':True}
print(json.dumps(proof),flush=True)
