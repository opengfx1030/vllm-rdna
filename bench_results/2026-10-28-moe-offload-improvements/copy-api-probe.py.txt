import ctypes as C, json, os, time, statistics
import torch
assert os.environ['HIP_VISIBLE_DEVICES']=='8'
torch.cuda.set_device(0)
print(json.dumps({'torch':torch.__version__,'hip':torch.version.hip,'device':torch.cuda.get_device_name(0),'arch':torch.cuda.get_device_properties(0).gcnArchName}),flush=True)
lib=C.CDLL(torch._C.__file__)
try: fn=lib.hipMemcpyBatchAsync
except AttributeError as e:
 print(json.dumps({'supported':False,'error':str(e)}),flush=True);raise SystemExit
fn.argtypes=[C.POINTER(C.c_void_p),C.POINTER(C.c_void_p),C.POINTER(C.c_size_t),C.c_size_t,C.c_void_p,C.c_void_p,C.c_size_t,C.POINTER(C.c_size_t),C.c_void_p]
fn.restype=C.c_int
sizes0=[1638400,51200,12800,819200,25600,6400]
stream=torch.cuda.Stream()
def descriptor(src,dst):
 n=len(src)
 return ((C.c_void_p*n)(*[x.data_ptr() for x in dst]),(C.c_void_p*n)(*[x.data_ptr() for x in src]),(C.c_size_t*n)(*[x.numel() for x in src]))
def batch(desc):
 fail=C.c_size_t(2**64-1)
 rc=fn(*desc,len(desc[0]),None,None,0,C.byref(fail),C.c_void_p(stream.cuda_stream))
 if rc: raise RuntimeError(f'hipMemcpyBatchAsync rc={rc} failIdx={fail.value}')
for misses in [1,4,16]:
 sizes=sizes0*misses
 src=[torch.empty(n,dtype=torch.uint8,pin_memory=True).fill_((i%100)+1) for i,n in enumerate(sizes)]
 dst=[torch.empty(n,dtype=torch.uint8,device='cuda') for n in sizes]
 desc=descriptor(src,dst)
 with torch.cuda.stream(stream):
  for x in dst:x.fill_(199)
  try:batch(desc)
  except RuntimeError as e:
   print(json.dumps({'supported':False,'misses':misses,'error':str(e)}),flush=True);raise SystemExit
  for x in dst:x.add_(1)
 stream.synchronize()
 exact=all(torch.equal(x.cpu(),s+1) for x,s in zip(dst,src))
 assert exact
 results={}
 for mode in ['six_copy_per_expert','batch_prebuilt','batch_rebuild']:
  enqueue=[];elapsed=[]
  for it in range(23):
   start=torch.cuda.Event(enable_timing=True);end=torch.cuda.Event(enable_timing=True)
   with torch.cuda.stream(stream):
    start.record();t=time.perf_counter_ns()
    if mode=='six_copy_per_expert':
     for d,s in zip(dst,src):d.copy_(s,non_blocking=True)
    else:batch(descriptor(src,dst) if mode=='batch_rebuild' else desc)
    cpu=(time.perf_counter_ns()-t)/1e3;end.record()
   end.synchronize()
   if it>=3:enqueue.append(cpu);elapsed.append(start.elapsed_time(end)*1000)
  assert all(torch.equal(x.cpu(),s) for x,s in zip(dst,src))
  results[mode]={'cpu_enqueue_us_median':statistics.median(enqueue),'event_us_median':statistics.median(elapsed),'iterations':20}
 print(json.dumps({'supported':True,'misses':misses,'bytes':sum(sizes),'host_plus_gpu_bytes':2*sum(sizes),'ordering_and_bytes_exact':exact,'results':results}),flush=True)
 del src,dst,desc
 torch.cuda.empty_cache()
