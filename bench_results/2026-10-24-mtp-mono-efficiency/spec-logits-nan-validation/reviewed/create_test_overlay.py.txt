"""Create a disposable Python package overlay for the focused NaN tests."""

import argparse
import hashlib
import json
import os
from pathlib import Path


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    ap = argparse.ArgumentParser(description=__doc__)
    for key in ("source", "patched-runner", "output"):
        ap.add_argument("--" + key, type=Path, required=True)
    ap.add_argument("--expected-patch-sha256", required=True)
    args = ap.parse_args()
    source = args.source.resolve(strict=True)
    output = args.output.absolute()
    relative = "vllm/v1/worker/gpu_model_runner.py"
    original = digest(source / relative)
    assert digest(args.patched_runner) == args.expected_patch_sha256
    assert not output.exists()
    excluded = {
        ".git",
        "build",
        ".deps",
        "bench_results",
        "__pycache__",
        ".cache",
        ".pytest_cache",
        "overlay_manifest.json",
    }

    def share(src, dst):
        dst.mkdir()
        for item in src.iterdir():
            if item.name in excluded:
                continue
            target = dst / item.name
            if item.is_dir() and not item.is_symlink():
                share(item, target)
            else:
                target.symlink_to(item.absolute())

    output.mkdir()
    for item in source.iterdir():
        if item.name in excluded:
            continue
        if item.name == "vllm":
            share(item, output / item.name)
        else:
            (output / item.name).symlink_to(
                item.absolute(), target_is_directory=item.is_dir()
            )
    target = output / relative
    assert target.is_symlink()
    target.unlink()
    with target.open("wb") as stream:
        stream.write(args.patched_runner.read_bytes())
        stream.flush()
        os.fsync(stream.fileno())
    assert digest(target) == args.expected_patch_sha256
    assert digest(source / relative) == original
    (output / "overlay_manifest.json").write_text(
        json.dumps(
            {
                "source": str(source),
                "output": str(output),
                "allowed_changes": [relative],
                "source_runner_sha256": original,
                "patched_runner_sha256": digest(target),
                "core_sha256": digest(output / "vllm/_rocm_C.abi3.so"),
                "scope": "Isolated Python tests; installed/serving files unchanged",
            },
            indent=2,
        )
        + "\n"
    )


if __name__ == "__main__":
    main()
