"""CPU-only curation of completed diagnostics; no raw replies or binaries."""
import hashlib
import importlib.util
import json
import re
from pathlib import Path

HERE=Path(__file__).resolve().parent
BASE=HERE.parent
RECORDS=[]


def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()


def copy(source,target=None,compact=False):
    source=BASE/source
    target=HERE/(target or str(source.relative_to(BASE)))
    target.parent.mkdir(parents=True,exist_ok=True)
    if compact:
        data=json.loads(source.read_text())
        target.write_text(json.dumps(data,separators=(',',':'),allow_nan=False)+'\n')
        assert json.loads(target.read_text())==data
    else:target.write_bytes(source.read_bytes())
    RECORDS.append({'source':str(source.relative_to(BASE)),'source_sha256':sha(source),'curated_path':str(target.relative_to(HERE)),'curated_sha256':sha(target),'transform':'JSON whitespace compacted; parsed payload identical' if compact else 'exact bytes'})


def derived(path,data,sources):
    output=HERE/path;output.parent.mkdir(parents=True,exist_ok=True)
    output.write_text(json.dumps(data,indent=2,allow_nan=False)+'\n')
    RECORDS.append({'curated_path':path,'curated_sha256':sha(output),'transform':'derived response-free subset/summary','sources':{s:sha(BASE/s) for s in sources}})


def journal(directory,target):
    before=json.loads((BASE/directory/'kernel.before.json').read_text())
    after=json.loads((BASE/directory/'kernel.after.json').read_text())
    spec=importlib.util.spec_from_file_location('curation_journal',BASE/'dense-mtp-failstop-serving-stage-v3/kernel_journal_v2.py')
    module=importlib.util.module_from_spec(spec);spec.loader.exec_module(module)
    additions=module.check_append_only(before,after)
    assert all(r.get('_TRANSPORT')=='kernel' and isinstance(r.get('__CURSOR'),str) and isinstance(r.get('MESSAGE'),str) for r in before+after)
    derived(target,{'status':'PASS','scope':'CPU recheck of archived privileged snapshots; no fresh journal read','before_records':len(before),'after_records':len(after),'new_records':additions,'new_faults':0,'append_only':True,'kernel_record_metadata_present':True},[directory+'/kernel.before.json',directory+'/kernel.after.json'])
