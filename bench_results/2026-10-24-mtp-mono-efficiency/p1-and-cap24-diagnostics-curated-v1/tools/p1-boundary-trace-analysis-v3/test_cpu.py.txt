#!/usr/bin/env python3
"""CPU checks for the Python-3.12-compatible libm fma adapter."""

import importlib.util
import math
import random
import unittest
from pathlib import Path

PATH = Path(__file__).with_name("analyze.py")
SPEC = importlib.util.spec_from_file_location("p1_analysis_v3", PATH)
assert SPEC is not None and SPEC.loader is not None
ANALYZER = importlib.util.module_from_spec(SPEC)
SPEC.loader.exec_module(ANALYZER)


class LibmFmaTests(unittest.TestCase):
    def test_elementary_fused_results(self):
        self.assertEqual(ANALYZER.C_FMA(1.5, 2.0, -3.0), 0.0)
        self.assertEqual(ANALYZER.C_FMA(0.5, 0.5, 0.25), 0.5)

    @unittest.skipUnless(hasattr(math, "fma"), "math.fma absent on this Python")
    def test_matches_python_math_fma(self):
        rng = random.Random(250252)
        for _ in range(1000):
            a, b, c = (rng.uniform(-20.0, 20.0) for _ in range(3))
            self.assertEqual(ANALYZER.C_FMA(a, b, c), math.fma(a, b, c))


if __name__ == "__main__":
    unittest.main()
