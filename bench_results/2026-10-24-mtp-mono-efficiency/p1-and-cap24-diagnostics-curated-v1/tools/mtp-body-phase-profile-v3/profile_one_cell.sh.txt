assert payload.get("status") == "PASS", payload

assert payload.get("reserved_hip_ids") == [6, 7, 8, 9], payload
PY
}

# Snapshot all four loaded worker mappings and their physical KFD GPU owners
# once the gate has launched the server. This is independent of HTTP metrics.
worker_snapshot() {
  for attempt in $(seq 1 360); do
    [[ $(cat /proc/sys/kernel/random/boot_id) == "$BOOT" ]] || return 2
    if [[ -f $RUNS_ROOT/summary.txt ]] && \
       grep -Fq 'READY in ' "$RUNS_ROOT/summary.txt" && \
       [[ -f $RUNS_ROOT/serve.log ]] && \
       [[ $(grep -Ec 'Worker_TP[0-3]_EP[0-9]+ pid=' "$RUNS_ROOT/serve.log" || true) -ge 4 ]]; then
      for snap_attempt in 1 2 3 4 5 6; do
        if "$VENV/bin/python" -S "$HERE/../core-inline-serving-ab/snapshot_workers.py" \
          --serve-log "$RUNS_ROOT/serve.log" --expected-tree "$CANDIDATE_TREE" \
          --expected-core-sha "$EXPECTED_SO" --output "$RUNS_ROOT/worker-provenance.json" \
          >> "$RUNS_ROOT/worker-provenance.log" 2>&1; then
          return 0
        fi
        sleep 2
      done
    fi
    sleep 10
  done
  echo 'worker provenance timeout waiting for four TP worker records' >&2
  return 1
}
