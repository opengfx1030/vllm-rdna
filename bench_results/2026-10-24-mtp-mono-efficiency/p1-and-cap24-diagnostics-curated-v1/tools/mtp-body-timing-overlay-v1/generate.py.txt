#!/usr/bin/env python3
"""Generate three Python-only diagnostics from exact healthy fail-stop sources."""

import hashlib
import json
from pathlib import Path

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[2]
PREFIX = "vllm/models/qwen4_exp/amd/mono/rdna2/"
BASE = {

    "layout.py": "00e730f8d40b5ec9966ff096040e872be2547068e60e4af87d93b8bb17a575c1",
    "k2.py": "64cc1b2a22737b59f2e6113ceeaa551509a63e4a307abe44c286849a426788dc",
}


def sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def replace(s, old, new):
    if s.count(old) != 1:
        raise ValueError(f"anchor count {s.count(old)}: {old[:70]}")
    return s.replace(old, new)


def transform(name, s):
    if name == "debug.py":
        s = replace(
            s,
            "STATS: dict[int, torch.Tensor] = {}",
            (
                "STATS: dict[int, torch.Tensor] = {}\n"
                "MTP_STATS: dict[int, torch.Tensor] = {}"
            ),
        )
        s = replace(s, "_last_log = [0.0]", "_last_log = [0.0]\n_mtp_last_log = [0.0]")
        s = replace(
            s,
            "def log_stats():",
            '''def statistics(scope: str = "model"):
    """Separate workspace stats; no automatic migration between scopes."""
    if scope == "model":
        return STATS
    if scope == "mtp":
        return MTP_STATS
    raise ValueError(f"unknown timing scope: {scope}")

    """

def main():
    manifest = {
        "diagnostic_only": True,
        "default_timing_false_gpu_extras": 0,
        "core_sha256": (
            "6facdbd60118fc14a5cf1ceb5c2494d38ed1be09f55e4636bb0f0825678e2eed"
        ),
        "changes": {},
    }
    for name, expected in BASE.items():
        path = ROOT / (PREFIX + name)
        if sha(path) != expected:
            raise ValueError(f"wrong exact healthy base: {path}")
        candidate = transform(name, path.read_text())
        target = HERE / "candidate-overlay" / (PREFIX + name)
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_text(candidate)
        manifest["changes"][PREFIX + name] = {
            "original": expected,
            "candidate": sha(target),
        }
    (HERE / "source-manifest.json").write_text(json.dumps(manifest, indent=2) + "\n")


if __name__ == "__main__":
    main()
